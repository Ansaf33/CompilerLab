decl
int factarr[10];
int fact(int n);

enddecl


int fact(int n){
  decl
  enddecl
  begin
  if( n == 0 ) then
    return 0;
  else if ( n == 1 ) then
    return 1;
    endif;
  endif;
  return n*fact(n-1);

  end;


}

int main(){
  decl
  int i;
  enddecl
  begin

  i = 0;
  while(i<10) do
    factarr[i] = fact(i);
    i = i + 1;
  endwhile;

  i = 0;
  while(i<10) do
    write(factarr[i]);

  endwhile;

  end;

}
